1: Peak Index in a Mountain Array (Leetcode 852)

class Solution {
public:
    int peakIndexInMountainArray(vector<int>& arr) {
        int low = 0, high = arr.size() - 1;
        while(low <= high)
        {
            int mid = low + (high - low) / 2;
            if(arr[mid] > arr[mid + 1] && arr[mid] > arr[mid - 1]) return mid;
            else if(arr[mid] > arr[mid + 1]) high = mid - 1;
            else low = mid + 1;
        }
        return -1;
    }
};

2: Find Minimum in Rotated Sorted Array (Leetcode 153)

class Solution {
public:
    int findMin(vector<int>& nums) {
        int low = 0, high = nums.size() - 1;
        while(low <= high)
        {
            int mid = (low + high) / 2;
            if(nums[mid] < nums[high]) high = mid;
            else low = mid + 1;
        }
        return nums[high];
    }
};

3: Search in Rotated Sorted Array (Leetcode 33)

class Solution {
public:
    int search(vector<int>& nums, int target) {
        int low = 0, high = nums.size() - 1;
        while(low <= high)
        {
            int mid = low + (high - low) / 2;
            if(nums[mid] == target) return mid;
            else if(nums[low] <= nums[mid])
            {
                if(nums[low] <= target && target < nums[mid]) high = mid - 1;
                else low = mid + 1;
            }
            else
            {
                if(nums[mid] < target && target <= nums[high]) low = mid + 1;
                else high = mid - 1;
            }
        }
        return -1;
    }
};

4: Kth Missing Positive Integer (Leetcode 1539)

class Solution {
public:
    int findKthPositive(vector<int>& arr, int k) {
        int low = 0, high = arr.size() - 1;
        int mid, ans = arr.size();

        while(low <= high)
        {
            mid = low + (high - low) / 2;
            if(arr[mid] - mid - 1 >= k)
            {
                ans = mid;
                high = mid - 1;
            }
            else low = mid + 1;
        }
        return k + ans;
    }
};

5: Find Peak Element (Leetcode 162)

class Solution {
public:
    int findPeakElement(vector<int>& nums) {
        int low = 0, high = nums.size() - 1, mid;
        while(low < high)
        {
            mid = (low + high) / 2;
            if(nums[mid] < nums[mid+1]) low = mid + 1;
            else high = mid;
        }
        return low;
    }
};

6: Special Array with X elements greater than or equal to X (Leetcode 1608)

class Solution {
public:
    int specialArray(vector<int>& nums) {
        int n = nums.size();
        int count;
        for(int i = 0; i <= n; ++i)
        {
            count = 0;
            for(int j = 0; j < n; ++j)
                if(nums[j] >= i) count++;
            
            if(count == i) return i;
        }
        return -1;
    }
};

7: Valid Perfect Square (Leetcode 367)

class Solution {
public:
    bool isPerfectSquare(int num) {
        int low = 0;
        int high = num/2 + 1;
        while(low <= high){
            long int mid = low + (high - low) / 2;
            if(mid*mid == num)
                return 1;
            else if(mid*mid < num)
                low = mid + 1;
            else
                high = mid - 1;
        }
        return 0;
    }
};
