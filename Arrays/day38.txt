1: Segregate 0's and 1's (GeekforGeeks)

class Solution {
  public:
    void segregate0and1(vector<int> &arr) {
        // code here
        int start = 0, end = arr.size() - 1;
        while(start < end) {
            while(start < end && arr[start] == 0) start++;
            while(start < end && arr[end] == 1) end--;
            swap(arr[start++], arr[end--]);
        }
    }
};

2: Two Sum II (Leetcode 167)

class Solution {
public:
    vector<int> twoSum(vector<int>& numbers, int target) {
        int start = 0, end = numbers.size() - 1;
        while(start < end) {
            if(numbers[start] + numbers[end] == target)
                return {start + 1, end + 1};
            else if(numbers[start] + numbers[end] < target) start++;
            else end--;
        }
        return {-1,-1};
    }
};

3: Pair With Given Difference (Interview Bit)

int Solution::solve(vector<int> &A, int B) {
    sort(A.begin(), A.end());
    int start = 0, end = 1, n = A.size();

    if(B < 0) B = -B;

    while(start < n && end < n) {
        if(start == end) {
            end++;
            continue;
        }

        int diff = A[end] - A[start];

        if(diff == B) return 1;
        else if(diff < B) end++;
        else start++;
    }

    return 0;
}

4: Product Pair (GeekforGeeks)

class Solution {
  public:
    bool isProduct(vector<int>& arr, long long target) {
        // code here
        long long prod;
        int start = 0, end = arr.size() - 1;
        while(start < end) {
            prod = arr[start] * arr[end];
            
            if(prod == target) return true;
            else if(prod < target) start++;
            else end--;
        }
        return false;
    }
};
