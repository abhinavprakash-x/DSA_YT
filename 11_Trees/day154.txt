1. Identical Trees (GeeksforGeeks)

class Solution {
  public:
    bool isIdentical(Node* r1, Node* r2) {
        // code here
        if(!r1 && !r2) return true;
        if(!r1 || !r2) return false;
        if(r1->data != r2->data) return false;
        
        bool c1, c2;
        
        c1 = isIdentical(r1->left, r2->left);
        c2 = isIdentical(r1->right, r2->right);
            
        return c1 && c2;
    }
};

2. Convert to Mirror Tree (GeeksforGeeks)

class Solution {
  public:
    void mirror(Node* root) {
        // code here
        if(!root) return;
        
        if(root) swap(root->left, root->right);
        
        mirror(root->left);
        mirror(root->right);
    }
};

3. Balanced Tree Check (GeeksforGeeks)

class Solution {
  public:
    int height(Node* root, bool &valid)
    {
        if(!root) return 0;
        
        int l_h = height(root->left, valid);
        int r_h = height(root->right, valid);
        
        if(abs(l_h - r_h) > 1) valid = false;
        
        return (1 + max(l_h, r_h));
    }
    
    bool isBalanced(Node* root) {
        // code here
        bool valid = true;
        height(root, valid);
        return valid;
    }
};

4. Zig Zag Traversal (GeeksforGeeks)

class Solution {
  public:
    vector<int> zigZagTraversal(Node* root) {
        // code here
        stack<Node*> s1, s2;
        vector<int> ans;
        
        if(!root) return ans;
        s2.push(root);
        
        while(!s1.empty() || !s2.empty())
        {
            while(!s1.empty())
            {
                Node* temp = s1.top();
                s1.pop();
                ans.push_back(temp->data);
                
                if(temp->right) s2.push(temp->right);
                if(temp->left) s2.push(temp->left);
            }
            
            while(!s2.empty())
            {
                Node* temp = s2.top();
                s2.pop();
                ans.push_back(temp->data);
                
                if(temp->left) s1.push(temp->left);
                if(temp->right) s1.push(temp->right);
            }
        }
        
        return ans;
    }
};

5. Check for Cousins in Binary Tree (GeeksforGeeks)

class Solution {
  public:
    bool areCousins(Node* root, int a, int b)
    {
        if(!root) return false;

        queue<pair<Node*, Node*>> q;
        q.push({root, NULL});

        while(!q.empty())
        {
            int lvl_size = q.size();

            Node* parent_a = NULL;
            Node* parent_b = NULL;

            for(int i = 0; i < lvl_size; i++)
            {
                Node* temp = q.front().first;
                Node* parent = q.front().second;
                q.pop();

                if(temp->data == a) parent_a = parent;
                if(temp->data == b) parent_b = parent;

                if(temp->left) q.push({temp->left, temp});

                if(temp->right) q.push({temp->right, temp});
            }

            if(parent_a && parent_b) return parent_a != parent_b;
            if(parent_a || parent_b) return false;
        }

        return false;
    }
};
