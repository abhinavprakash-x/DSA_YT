1. Left View of Binary Tree (GeeksforGeeks)

class Solution {
  public:
    vector<int> leftView(Node *root) {
        // code here
        if(!root) return {};
        
        queue<Node*> q;
        vector<int> ans;
        q.push(root);
        
        while(!q.empty())
        {
            int n = q.size();
            ans.push_back(q.front()->data);
            
            for(int i = 0; i < n; ++i)
            {
                Node* temp = q.front();
                q.pop();
                
                if(temp->left) q.push(temp->left);
                if(temp->right) q.push(temp->right);
            }
        }
        return ans;
    }
};

3. Top View of Binary Tree (GeeksforGeeks)

class Solution {
  public:
    vector<int> topView(Node *root) {
        if(!root) return {};

        map<int, int> mp;
        queue<pair<Node*, int>> q;

        q.push({root, 0});

        while(!q.empty())
        {
            auto [temp, hd] = q.front();
            q.pop();

            if(mp.find(hd) == mp.end()) mp[hd] = temp->data;

            if(temp->left) q.push({temp->left, hd - 1});
            if(temp->right) q.push({temp->right, hd + 1});
        }

        vector<int> ans;
        for(auto it : mp) ans.push_back(it.second);

        return ans;
    }
};
