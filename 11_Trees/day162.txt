// Fixed
class Solution {
  public:
    bool find(Node* root, int target, vector<Node*>& path) {
        if(!root) return false;
        path.push_back(root);

        if(root->data == target) return true;

        if(find(root->left, target, path) ||
            find(root->right, target, path)) {
            return true;
        }

        path.pop_back();
        return false;
    }

    int height(Node* root) {
        if (!root) return -1;
        return 1 + max(height(root->left), height(root->right));
    }

    int minTime(Node* root, int target) {
        vector<Node*> path;
        if(!find(root, target, path)) return 0;
        int n = path.size();

        int ans = height(path[n - 1]);

        for (int i = n - 2; i >= 0; --i)
        {
            Node* ancestor = path[i];
            Node* childOnPath = path[i + 1];

            Node* opposite;

            if(ancestor->left == childOnPath) opposite = ancestor->right;
            else opposite = ancestor->left;

            int dist = n - 1 - i;
            int time = dist + 1 + height(opposite);
            ans = max(ans, time);
        }
        return ans;
    }
};

// Video
class Solution {
  public:
    int Burn(Node* root, int& timer, int target)
    {
        if(!root) return 0;
        if(root->data == target) return -1;

        int Left = Burn(root->left, timer, target);
        int Right = Burn(root->right, timer, target);

        if(Left < 0)
        {
            timer = max(timer, abs(Left) + Right);
            return Left - 1;
        }

        if(Right < 0)
        {
            timer = max(timer, Left + abs(Right));
            return Right - 1;
        }

        return 1 + max(Left, Right);
    }

    Node* find(Node* root, int target)
    {
        if (!root) return nullptr;

        if (root->data == target)
            return root;

        Node* l = find(root->left, target);
        if (l) return l;

        return find(root->right, target);
    }

    int Height(Node* root) {
        if(!root) return 0;
        return 1 + max(Height(root->left), Height(root->right));
    }

    int minTime(Node* root, int target) {
        if(!root) return 0;

        int timer = 0;
        Burn(root, timer, target);

        Node* BurnNode = find(root, target);
        int high = Height(BurnNode) - 1;

        return max(timer, high);
    }
};

// Correct Chat GPT
class Solution{
public:
    int pathSum(Node* root, int& sum, bool& found)
    {
        if(!root) return 0;
        if(!root->left && !root->right) return root->data;

        int l = pathSum(root->left, sum, found);
        int r = pathSum(root->right, sum, found);

        if(root->left && root->right)
        {
            found = true;
            sum = max(sum, l + root->data + r);
            return root->data + max(l, r);
        }

        if(root->left) return root->data + l;
        return root->data + r;
    }

    int maxPathSum(Node* root)
    {
        int sum = INT_MIN;
        bool found = false;

        pathSum(root, sum, found);

        return found ? sum : -1;
    }
};
