Q1: Search a 2D Matrix (Leetcode 74)

class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int m = matrix.size();
        int n = matrix[0].size();
        int start = 0, end = (m * n) - 1;

        while(start <= end) {
            int mid = start + (end - start) / 2;
            int row_idx = mid / n;
            int col_idx = mid % n;

            if(matrix[row_idx][col_idx] == target) return true;
            else if(matrix[row_idx][col_idx] < target) start = mid + 1;
            else end = mid - 1;
        }
        return false;
    }
};

Q3: Count zeros in a sorted matrix (GeekforGeeks)

class Solution {
  public:
    int countZeros(vector<vector<int>>& mat) {
        // code here
        int n = mat.size();
        int count = 0;
        int i = 0, j = n - 1;
        
        while(i < n && j < n) {
            if(mat[i][j] == 0) {
                count += (j + 1);
                i++;
            }
            else {
                j--;
            }
        }
        return count;
    }
};

Q4: Row with Max 1s in Rowwise Sorted (GeekforGeeks)

class Solution {
  public:
    int rowWithMax1s(vector<vector<int>> &arr) {
        // code here
        int n = arr.size();
        int ans = -1;
        int max = 0;
        
        for(int i = 0; i < n; ++i) {
            int count = 0;
            
            for(int j = 0; j < n; ++j) {
                if(arr[i][j] == 1) count++;
            }
            
            if(max < count) {
                max = count;
                ans = i;
            }
        }
        return ans;
    }
};
