Q1. Queue using Array (GeeksforGeeks)

TextBook Level Implementation (but fails on GFG)
class myQueue {
  private:
    vector<int> q;
    int front;
    int rear;
    int size;
    
  public:
    myQueue(int n) {
        // Define Data Structures
        q.resize(n);
        front = -1;
        rear = -1;
        size = n;
    }

    bool isEmpty() {
        // check if the queue is empty
        return front == -1;
    }

    bool isFull() {
        // check if the queue is full
        return rear == size - 1;
    }

    void enqueue(int x) {
        // Adds an element x at the rear of the queue.
        if(isFull()) return;
        
        if(isEmpty()) front = 0;
        rear++;
        q[rear] = x;
    }

    void dequeue() {
        // Removes the front element of the queue.
        if(isEmpty()) return;
        
        if(front == rear)
        {
            front = -1;
            rear = -1;
        }
        else front++;
    }

    int getFront() {
        // Returns the front element of the queue.
        return isEmpty() ? -1 : q[front];
    }

    int getRear() {
        // Return the last element of queue
        return isEmpty() ? -1 : q[rear];
    }
};

GFG Solution, replace dequeue with this
void dequeue() {
    if(isEmpty()) return;

    for(int i = front; i < rear; i++)
        q[i] = q[i + 1];

    rear--;

    if(rear == -1) front = -1;
}

Q2. Queue using Linked List (GeeksforGeeks)

class Node {
public:
    int data;
    Node* next;

    Node(int val) {
        data = val;
        next = nullptr;
    }
};

class myQueue {
private:
    Node *front, *rear;
    int count;

public:
    myQueue() {
        front = rear = nullptr;
        count = 0;
    }

    bool isEmpty() {
        return front == nullptr;
    }

    void enqueue(int x) {
        Node* temp = new Node(x);

        if (isEmpty()) {
            front = rear = temp;
        } else {
            rear->next = temp;
            rear = temp;
        }

        count++;
    }

    void dequeue() {
        if (isEmpty()) return;

        Node* temp = front;
        front = front->next;
        delete temp;

        count--;

        if (front == nullptr) {
            rear = nullptr;
        }
    }

    int getFront() {
        return isEmpty() ? -1 : front->data;
    }

    int size() {
        return count;
    }
};
