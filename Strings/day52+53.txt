Q1: Longest Pallindrome (Leetcode 409)

class Solution {
public:
    int longestPalindrome(string s) {
        vector<int> lower(26,0), upper(26,0);

        for(char c : s) {
            if (c >= 'a') lower[c - 'a']++;
            else upper[c - 'A']++;
        }

        int count = 0, odd = 0;
        for(int i = 0; i < 26; ++i) {
            if(lower[i] % 2 == 0) count += lower[i];
            else {
                odd = 1;
                count += lower[i] - 1;
            }
            if(upper[i] % 2 == 0) count += upper[i];
            else {
                odd = 1;
                count += upper[i] - 1;
            }
        }

        if(odd) count++;
        return count;
    }
};

Q2: Sorting the Sentence (Leetcode 1859)

class Solution {
public:
    string sortSentence(string s) {
        vector<string> ans(9);
        string temp;
        
        for(char c : s) {
            if(c == ' ') {
                int pos = temp[temp.size() - 1] - '0';
                temp.pop_back();
                ans[pos - 1] = temp;
                temp.clear();
            }
            else {
                temp += c;
            }
        }
        int pos = temp[temp.size() - 1] - '0';
        temp.pop_back();
        ans[pos - 1] = temp;
        temp.clear();

        for(int i = 0; i < 9; ++i) {
            if(ans[i] != "") {
                temp.append(ans[i]);
                temp.append(" ");
            }
        }
        temp.pop_back();

        return temp;
    }
};
