class Solution {
  public:
    Node* merge_sorted(Node *head1, Node *head2)
    {
        if(!head1) return head2;
        if(!head2) return head1;
        
        Node dummy(0);
        Node *ptr = &dummy;
        Node *ptr1 = head1, *ptr2 = head2;
        
        while(ptr1 && ptr2)
        {
            if(ptr1->data <= ptr2->data)
            {
                ptr->bottom = ptr1;
                ptr1 = ptr1->bottom;
            }
            else
            {
                ptr->bottom = ptr2;
                ptr2 = ptr2->bottom;
            }
            ptr = ptr->bottom;
        }
        
        ptr->bottom = ptr1 ? ptr1 : ptr2;
        return dummy.bottom;
    }
  
    Node* flatten(Node* head) {
        // code here
        if (!head) return nullptr;

        Node *ptr1 = head;
        while(ptr1->next != nullptr)
        {
            Node *ptr2 = ptr1->next;
            Node *ptr3 = ptr2->next;
            
            ptr1 = merge_sorted(ptr1, ptr2);
            ptr1->next = ptr3;
        }
        
        return ptr1;
    }
};

                list1 = list1->next;
            }
            else
            {
                ptr->next = list2;
                list2 = list2->next;
            }
            ptr = ptr->next;
        }
        ptr->next = (list1 != nullptr) ? list1 : list2;
        return merge_list.next;
    }

    Node* mergeKLists(vector<Node*>& lists) {
        if(lists.empty()) return nullptr;

        for(int i = 0; i < lists.size() - 1; ++i)

        }
        return lists[lists.size() - 1];
    }
};

class Solution {
public:
    Node* merge(Node* list1, Node* list2)
    {
        Node *head = new Node(0);
        Node *tail = head;
        
        while(list1 && list2)
        {
            if(list1->data <= list2->data)
            {
                tail->next = list1;
                list1 = list1->next;
                tail = tail->next;
                tail->next = nullptr;
            }
            else
            {
                tail->next = list2;
                list2 = list2->next;
                tail = tail->next;
                tail->next = nullptr;
            }
        }
        
        tail->next = list1 ? list1 : list2;
        return head->next;
    }
    
    void merge_sort(vector<Node*>& lists, int start, int end)
    {
        if(start >= end) return;
        
        int mid = start + (end - start) / 2;
        merge_sort(lists, start, mid);
        merge_sort(lists, mid + 1, end);
        
        lists[start] = merge(lists[start], lists[mid + 1]);
    }

    Node* mergeKLists(vector<Node*>& lists) {
        if(lists.empty()) return nullptr;
        
        merge_sort(lists, 0, lists.size() - 1);
        
        return lists[0];
    }
};

Q3. Merge Sort for Linked List (GeekforGeeks)

class Solution {
public:
    Node* merge(Node* head1, Node* head2)
    {
        Node dummy(0);
        Node* tail = &dummy;

        while (head1 && head2)
        {
            if (head1->data <= head2->data)
            {
                tail->next = head1;
                head1 = head1->next;
            }
            else
            {
                tail->next = head2;
                head2 = head2->next;
            }

            tail = tail->next;
        }

        tail->next = head1 ? head1 : head2;
        return dummy.next;
    }

    Node* mergeSort(Node* head)
    {
        if (!head || !head->next) return head;

        Node* slow = head;
        Node* fast = head->next;

        while (fast && fast->next)
        {
            slow = slow->next;
            fast = fast->next->next;
        }
        Node* right = slow->next;
        slow->next = nullptr;

        Node* left = mergeSort(head);
        right = mergeSort(right);

        return merge(left, right);
    }
};

Q4. Merge Sort for Doubly Linked List (GeekforGeeks)

class Solution {
public:

    Node* merge(Node* head1, Node* head2)
    {
        Node dummy(0);
        Node* tail = &dummy;

        while (head1 && head2)
        {
            if (head1->data <= head2->data)
            {
                tail->next = head1;
                head1->prev = tail;
                head1 = head1->next;
            }
            else
            {
                tail->next = head2;
                head2->prev = tail;
                head2 = head2->next;
            }

            tail = tail->next;
        }

        if(head1)
        {
            tail->next = head1;
            head1->prev = tail;
        }
        else
        {
            tail->next = head2;
            if(head2) head2->prev = tail;
        }

        Node* newHead = dummy.next;
        newHead->prev = nullptr;
        return newHead;
    }

    Node* mergeSort(Node* head)
    {
        if (!head || !head->next)
            return head;

        // Find middle
        Node* slow = head;
        Node* fast = head->next;

        while (fast && fast->next)
        {
            slow = slow->next;
            fast = fast->next->next;
        }

        // Split into two lists
        Node* right = slow->next;
        slow->next = nullptr;

        // Sort both halves
        Node* left = mergeSort(head);
        right = mergeSort(right);

        // Merge
        return merge(left, right);
    }
};

Q5. Reorder List (GeekforGeeks)

class Solution {
  public:
    Node* reverse(Node *head) {
        Node *prev = nullptr, *curr = head, *next = nullptr;
        while(curr != nullptr) {
            next = curr->next;
            curr->next = prev;
            prev = curr;
            curr = next;
        }
        return prev;
    }
  
    void reorderList(Node* head) {
        // code here
        Node *slow = head, *fast = head;
        
        while(fast && fast->next)
        {
            slow = slow->next;
            fast = fast->next->next;
        }
        
        Node *temp = slow->next;
        slow->next = nullptr;
        
        temp = reverse(temp);
        
        Node dummy(0);
        Node *ptr = &dummy;
        
        while(head && temp)
        {
            ptr->next = head;
            head = head->next;
            ptr = ptr->next;
            
            ptr->next = temp;
            temp = temp->next;
            ptr = ptr->next;
        }
        
        ptr->next = temp ? temp : head;
        head = dummy.next;
    }
};
