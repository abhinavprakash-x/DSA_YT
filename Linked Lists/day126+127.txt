class Solution {
  public:
    Node* merge_sorted(Node *head1, Node *head2)
    {
        if(!head1) return head2;
        if(!head2) return head1;
        
        Node dummy(0);
        Node *ptr = &dummy;
        Node *ptr1 = head1, *ptr2 = head2;
        
        while(ptr1 && ptr2)
        {
            if(ptr1->data <= ptr2->data)
            {
                ptr->bottom = ptr1;
                ptr1 = ptr1->bottom;
            }
            else
            {
                ptr->bottom = ptr2;
                ptr2 = ptr2->bottom;
            }
            ptr = ptr->bottom;
        }
        
        ptr->bottom = ptr1 ? ptr1 : ptr2;
        return dummy.bottom;
    }
  
    Node* flatten(Node* head) {
        // code here
        if (!head) return nullptr;

        Node *ptr1 = head;
        while(ptr1->next != nullptr)
        {
            Node *ptr2 = ptr1->next;
            Node *ptr3 = ptr2->next;
            
            ptr1 = merge_sorted(ptr1, ptr2);
            ptr1->next = ptr3;
        }
        
        return ptr1;
    }
};

            else

                ptr->next = list2;

        }
        return lists[lists.size() - 1];
    }

        {

                tail = tail->next;
                tail->next = nullptr;
            }
            else
            {
                tail->next = list2;
                list2 = list2->next;
                tail = tail->next;
                tail->next = nullptr;
            }
        }
        
        tail->next = list1 ? list1 : list2;
        return head->next;
    }
    
    void merge_sort(vector<Node*>& lists, int start, int end)
    {
        if(start >= end) return;
        
        int mid = start + (end - start) / 2;
        merge_sort(lists, start, mid);
        merge_sort(lists, mid + 1, end);
        
        lists[start] = merge(lists[start], lists[mid + 1]);
    }

    Node* mergeKLists(vector<Node*>& lists) {
        if(lists.empty()) return nullptr;
        
        merge_sort(lists, 0, lists.size() - 1);
        
        return lists[0];
    }
};

Q3. Merge Sort for Linked List (GeekforGeeks)

class Solution {
public:
    Node* merge(Node* head1, Node* head2)
    {
        Node dummy(0);
        Node* tail = &dummy;

        while (head1 && head2)
        {
            if (head1->data <= head2->data)
            {
                tail->next = head1;
                head1 = head1->next;
            }
            else
            {
                tail->next = head2;
                head2 = head2->next;
            }

            tail = tail->next;
        }

        tail->next = head1 ? head1 : head2;
        return dummy.next;
    }

    Node* mergeSort(Node* head)
    {
        if (!head || !head->next) return head;

        Node* slow = head;
        Node* fast = head->next;

        while (fast && fast->next)
        {
            slow = slow->next;
            fast = fast->next->next;
        }
        Node* right = slow->next;
        slow->next = nullptr;

        Node* left = mergeSort(head);
        right = mergeSort(right);

        return merge(left, right);
    }
};

Q4. Merge Sort for Doubly Linked List (GeekforGeeks)

class Solution {
public:

    Node* merge(Node* head1, Node* head2)
    {
        Node dummy(0);
        Node* tail = &dummy;

        while (head1 && head2)
        {
            if (head1->data <= head2->data)
            {
                tail->next = head1;
                head1->prev = tail;
                head1 = head1->next;
            }
            else
            {
                tail->next = head2;
                head2->prev = tail;
                head2 = head2->next;
            }

            tail = tail->next;
        }

        if(head1)
        {
            tail->next = head1;
            head1->prev = tail;
        }
        else
        {
            tail->next = head2;
            if(head2) head2->prev = tail;
        }

        Node* newHead = dummy.next;
        newHead->prev = nullptr;
        return newHead;
    }

    Node* mergeSort(Node* head)
    {
        if (!head || !head->next)
            return head;

        // Find middle
        Node* slow = head;
        Node* fast = head->next;

        while (fast && fast->next)
        {
            slow = slow->next;
            fast = fast->next->next;
        }

        // Split into two lists
        Node* right = slow->next;
        slow->next = nullptr;

        // Sort both halves
        Node* left = mergeSort(head);
        right = mergeSort(right);

        // Merge
        return merge(left, right);
    }
};

Q5. Reorder List (GeekforGeeks)

class Solution {
  public:
    Node* reverse(Node *head) {
        Node *prev = nullptr, *curr = head, *next = nullptr;
        while(curr != nullptr) {
            next = curr->next;
            curr->next = prev;
            prev = curr;
            curr = next;
        }
        return prev;
    }
  
    void reorderList(Node* head) {
        // code here
        Node *slow = head, *fast = head;
        
        while(fast && fast->next)
        {
            slow = slow->next;
            fast = fast->next->next;
        }
        
        Node *temp = slow->next;
        slow->next = nullptr;
        
        temp = reverse(temp);
        
        Node dummy(0);
        Node *ptr = &dummy;
        
        while(head && temp)
        {
            ptr->next = head;
            head = head->next;
            ptr = ptr->next;
            
            ptr->next = temp;
            temp = temp->next;
            ptr = ptr->next;
        }
        
        ptr->next = temp ? temp : head;
        head = dummy.next;
    }
};

Q6. Clone List with Next and Random (GeekforGeeks)

/* Structure of Linked List Node

        data = x;

class Solution {
  public:
    Node* cloneLinkedList(Node* head) {
        // code here
        Node *ptr = head;
        Node clone(0);
        Node *clone_ptr = &clone;
        
        while(ptr != nullptr)
        {
            clone_ptr->next = new Node(ptr->data);
            ptr = ptr->next;
            clone_ptr = clone_ptr->next;
        }
        clone_ptr->next = nullptr;
        
        ptr = head;
        clone_ptr = clone.next;
        
        while(ptr != nullptr)
        {
            Node *temp = head;
            int distance = 0;
            
            while(temp != ptr->random)
            {
                temp = temp->next;
                distance++;
            }
            distance++;
            
            temp = clone.next;
            while(--distance) temp = temp->next;
            
            clone_ptr->random = temp;
            
            ptr = ptr->next;
            clone_ptr = clone_ptr->next;
        }
        
        return clone.next;
    }
};

        clone_ptr = clone.next;

// Time O(n) Space O(1)
class Solution {
public:
    Node* cloneLinkedList(Node* head) {
        if(!head) return nullptr;

        // 1. Insert cloned node after every original node
        Node* curr = head;

        while(curr)
        {
            Node* clone = new Node(curr->data);

            clone->next = curr->next;
            curr->next = clone;

            curr = clone->next;
        }

        // 2. Set random pointers of cloned nodes
        curr = head;
        while(curr)
        {
            Node* clone = curr->next;

            if(curr->random) clone->random = curr->random->next;
            else clone->random = nullptr;

            curr = clone->next;
        }

        // 3. Separate original and cloned lists
        curr = head;
        Node* cloneHead = head->next;
        Node* cloneCurr = cloneHead;

        while (curr)
        {
            curr->next = cloneCurr->next;
            curr = curr->next;

            if (curr)
            {
                cloneCurr->next = curr->next;
                cloneCurr = cloneCurr->next;
            }
        }

        return cloneHead;
    }
};

Q7. Partition a Linked List around a given value (GeekforGeeks)

class Solution {
public:
    Node* partition(Node* head, int x) {

        Node small(0);
        Node equal(0);
        Node large(0);

        Node* smallTail = &small;
        Node* equalTail = &equal;
        Node* largeTail = &large;

        Node* curr = head;

        while (curr)
        {

            Node* next = curr->next;

            if (curr->data < x)
            {
                smallTail->next = curr;
                smallTail = smallTail->next;
            }
            else if (curr->data == x)
            {
                equalTail->next = curr;
                equalTail = equalTail->next;
            }
            else
            {
                largeTail->next = curr;
                largeTail = largeTail->next;
            }

            curr = next;
        }
        smallTail->next = equal.next ? equal.next : large.next;
        equalTail->next = large.next ? large.next : nullptr;
        largeTail->next = nullptr;

        return small.next;
    }
};

Q8. Subtraction in a Linked List (GeekforGeeks)

class Solution {
public:
    Node* reverse(Node* head)
    {
        Node* prev = nullptr;
        Node* curr = head;

        while (curr)
        {
            Node* next = curr->next;

            curr->next = prev;
            prev = curr;
            curr = next;
        }

        return prev;
    }

    long long length(Node* head)
    {
        long long size = 0;

        while (head)
        {
            size++;
            head = head->next;
        }

        return size;
    }

    Node* subLinkedList(Node* head1, Node* head2)
    {
        long long len1 = length(head1);
        long long len2 = length(head2);

        Node* larger;
        Node* smaller;

        if (len1 > len2)
        {
            larger = head1;
            smaller = head2;
        }
        else if (len1 < len2)
        {
            larger = head2;
            smaller = head1;
        }
        else
        {
            Node* p1 = head1;
            Node* p2 = head2;

            while(p1 && p1->data == p2->data)
            {
                p1 = p1->next;
                p2 = p2->next;
            }
            // both list are same numbers
            if (!p1) return new Node(0);

            if (p1->data > p2->data)
            {
                larger = head1;
                smaller = head2;
            }
            else
            {
                larger = head2;
                smaller = head1;
            }
        }

        larger = reverse(larger);
        smaller = reverse(smaller);

        Node dummy(0);
        Node* tail = &dummy;

        Node* ptr1 = larger;
        Node* ptr2 = smaller;

        int borrow = 0;

        while (ptr1)
        {
            int x = ptr1->data - borrow;
            int y = ptr2 ? ptr2->data : 0;

            if (x < y)
            {
                x += 10;
                borrow = 1;
            }
            else
            {
                borrow = 0;
            }

            tail->next = new Node(x - y);
            tail = tail->next;

            ptr1 = ptr1->next;
            if(ptr2) ptr2 = ptr2->next;
        }
        
        Node* result = reverse(dummy.next);
        while (result && result->data == 0 && result->next) result = result->next;

        return result;
    }
};

Extra Question: Swap Nodes in Pairs (Leetcode 24)

class Solution {
public:
    ListNode* swapPairs(ListNode* head) {

        if (!head || !head->next)
            return head;

        ListNode dummy(0);
        dummy.next = head;

        ListNode* prev = &dummy;
        ListNode* curr = head;
        ListNode* next = head->next;

        while (curr && next) {

            // Connect previous part to second node
            prev->next = next;

            // Put first node after second
            curr->next = next->next;
            next->next = curr;

            // Move to next pair
            prev = curr;
            curr = curr->next;

            if (curr)
                next = curr->next;
        }

        return dummy.next;
    }
};
