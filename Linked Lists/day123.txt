Q1. Add Number Linked List (GeekforGeeks)

class Solution {
  public:
    Node* reverse(Node* head) {
        Node *curr = head, *prev, *next;
        prev = next = nullptr;
        
        while(curr != nullptr) {
            next = curr->next;
            curr->next = prev;
            prev = curr;
            curr = next;
        }
        
        return prev;
    }
  
    Node* addTwoLists(Node* head1, Node* head2) {
        // code here
        head1 = reverse(head1);
        head2 = reverse(head2);
        
        Node *ptr1 = head1, *ptr2 = head2;
        Node *ptr3 = nullptr, *result_head = nullptr;
        
        int carry = 0, sum = 0;
        bool first = true;
        
        while(ptr1 || ptr2 || carry)
        {
            int num1 = (ptr1) ? ptr1->data : 0;
            int num2 = (ptr2) ? ptr2->data : 0;
            
            sum = num1 + num2 + carry;
            if(sum > 9)
            {
                carry = 1;
                sum -= 10;
            }
            else carry = 0;
            
            if(first)
            {
                first = false;
                ptr3 = new Node(sum);
                result_head = ptr3;
            }
            else
            {
                ptr3->next = new Node(sum);
                ptr3 = ptr3->next;
            }
            
            if(ptr1) ptr1 = ptr1->next;
            if(ptr2) ptr2 = ptr2->next;
        }
        
        result_head = reverse(result_head);
        
        while(result_head->next && result_head->data == 0) result_head = result_head->next;
        return result_head;
    }
};

Q2. Add Two Numbers (Leetcode 2)

class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode result(0);
        ListNode* ptr3 = &result;
        int carry = 0;
        while (l1 || l2 || carry) {
            int x = (l1 ? l1->val : 0);
            int y = (l2 ? l2->val : 0);
            int sum = x + y + carry;
            carry = sum / 10;
            ptr3->next = new ListNode(sum % 10);
            ptr3 = ptr3->next;
            if (l1) l1 = l1->next;
            if (l2) l2 = l2->next;
        }
        return result.next;
    }
};

Q3. Linked List Group Reverse (GeekforGeeks)

class Solution {
  public:
    Node* reverse(Node *head) {
        Node *prev = nullptr, *curr = head, *next = nullptr;
        while(curr != nullptr) {
            next = curr->next;
            curr->next = prev;
            prev = curr;
            curr = next;
        }
        return prev;
    }

    Node *reverseKGroup(Node *head, int k) {
        // code here
        if(k == 1) return head;
        if(head->next == nullptr) return head;

        Node *ptr = head, *temp_head = head, *temp, *prev_tail = nullptr;
        int count = 1;
        bool first = true;

        while(ptr != nullptr)
        {
            while(count != k && ptr->next != nullptr)
            {
                count++;
                ptr = ptr->next;
            }

            temp = ptr->next;
            ptr->next = nullptr;

            Node *old_head = temp_head;
            temp_head = reverse(temp_head);

            if(first)
            {
                first = false;
                head = temp_head;
            }
            else prev_tail->next = temp_head;

            old_head->next = temp;
            prev_tail = old_head;

            temp_head = temp;
            ptr = temp;
            count = 1;
        }

        return head;
    }
};

Q4. Reverse Nodes in k-Group (Leetcode 25)

class Solution {
public:
    ListNode* reverse(ListNode* head) {
        ListNode* prev = nullptr, *curr = head;
        while (curr)
        {
            ListNode* next = curr->next;
            curr->next = prev;
            prev = curr;
            curr = next;
        }
        return prev;
    }

    ListNode* reverseKGroup(ListNode* head, int k) {
        if (!head || k == 1) return head;

        ListNode dummy(0);
        dummy.next = head;

        ListNode* prev_tail = &dummy;
        ListNode* temp_head = head;

        while (temp_head) {
            // Check whether k nodes exist
            ListNode* ptr = temp_head;

            for (int i = 1; i < k; i++) {
                if (!ptr->next) return dummy.next;
                ptr = ptr->next;
            }

            ListNode* temp = ptr->next;
            ptr->next = nullptr;

            ListNode* old_head = temp_head;
            temp_head = reverse(temp_head);
            prev_tail->next = temp_head;

            old_head->next = temp;
            prev_tail = old_head;
            temp_head = temp;
        }

        return dummy.next;
    }
};

Q5. Reverse Sublist of Linked List (GeekforGeeks)

class Solution {
  public:
    void reverse(Node *head) {
        Node *prev = nullptr, *curr = head, *next = nullptr;
        while(curr != nullptr)
        {
            next = curr->next;
            curr->next = prev;
            prev = curr;
            curr = next;
        }
    }

    Node* reverseBetween(int a, int b, Node* head) {
        // code here
        Node *prev_a = nullptr, *a_ptr = head, *next_b;
        Node *b_ptr;
        int count = 1;

        while(count != a)
        {
            prev_a = a_ptr;
            a_ptr = a_ptr->next;
            count++;
        }
        b_ptr = a_ptr;

        while(count != b)
        {
            b_ptr = b_ptr->next;
            count++;
        }
        next_b = b_ptr->next;
        b_ptr->next = nullptr;

        reverse(a_ptr);

        if(prev_a) prev_a->next = b_ptr;
        else head = b_ptr;

        a_ptr->next = next_b;

        return head;
    }
};

Q6. Multiply Two Linked Lists (GeekforGeeks)

class Solution {
  public:
    int get_num(Node *num) {
        int n = 0;
        
        while(num)
        {
            n = n * 10 + num->data;
            num = num->next;
        }
        
        return n;
    }
  
    int multiplyTwoLists(Node* first, Node* second) {
        // code here
        long long num1 = get_num(first);
        long long num2 = get_num(second);
        
        return static_cast<int>((num1 * num2) % 1000000007);
    }
};
