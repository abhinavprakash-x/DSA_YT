Q1. Insertion in a Doubly Linked List (GeekforGeeks)

class Solution {
  public:
    Node* insertAtPos(Node* head, int p, int x) {
        // code here
        int pos = 0;
        Node* temp = head;
        
        while(pos != p && temp != nullptr)
        {
            pos++;
            temp = temp->next;
        }
        Node* ptr = new Node(x);
        ptr->next = temp->next;
        ptr->prev = temp;
        if(temp->next != nullptr) temp->next->prev = ptr;
        temp->next = ptr;
        
        return head;
    }
};

Q2. Reverse a Doubly Linked List (GeekforGeeks)

class Solution {
  public:
    Node *reverse(Node *head) {
        // code here
        Node* curr = head;
        Node* prev = nullptr;
        while(curr != nullptr)
        {
            prev = curr->prev;
            curr->prev = curr->next;
            curr->next = prev;
            curr = curr->prev;
        }
        return prev->prev;
    }
};
