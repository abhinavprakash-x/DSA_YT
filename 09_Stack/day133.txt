Q1. Implement Stack Using Array (GeekforGeeks)

class myStack {
    vector<int> stack;
    int top;
    int size;
    
  public:
    myStack(int n) {
        stack.resize(n);
        top = -1;
        size = n - 1;
    }

    bool isEmpty() {
        return top == -1;
    }

    bool isFull() {
        return top == size;
    }

    void push(int x) {
        if(isFull()) return;
        top++;
        stack[top] = x;
    }

    void pop() {
        if(isEmpty()) return;
        stack[top] = 0;
        top--;
    }

    int peek() {
        if(isEmpty()) return -1;
        return stack[top];
    }
};

Q2. Stack Using Linked List (GeekforGeeks)

/* Structure of linked list Node
class Node {
  public:
    int data;
    Node* next;

    Node(int val) {
        data = val;
        next = nullptr;
    }
};*/
class myStack {
    Node *head;
    int s_size;

  public:
    myStack() {
        head = nullptr;
        s_size = 0;
    }

    bool isEmpty() {
        return head == nullptr;
    }

    void push(int x) {
        Node *ptr = new Node(x);
        ptr->next = head;
        head = ptr;
        
        s_size++;
    }

    void pop() {
        if (head == nullptr) return;
        
        Node *temp = head;
        head = head->next;
        
        delete temp;
        s_size--;
    }

    int peek() {
        if(head == nullptr) return -1;
        return head->data;
    }

    int size() {
        return s_size;
    }
};
