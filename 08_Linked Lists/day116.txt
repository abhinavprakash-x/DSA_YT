Q1. Remove Nth Node From End of List (Leetcode 19)

class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        if (!head) return nullptr;
        ListNode* PTR = head;
        int len = 1;
        while(PTR->next != nullptr)
        {
            PTR = PTR->next;
            ++len;
        }
        int idx = len - n;
        if(idx == 0)
        {
            ListNode* temp = head;
            head = head->next;
            delete temp;
            return head;
        }
        PTR = head;
        ListNode* PREPTR = nullptr;
        for(int i = 0; i < idx; ++i)
        {
            PREPTR = PTR;
            PTR = PTR->next;
        }
        PREPTR->next = PTR->next;
        delete PTR;
        return head;
    }
};

Q2. Remove Every k'th in Linked List (GeekforGeeks)

class Solution {
  public:
    Node* deleteK(Node* head, int K) {
        // code here
        if(K == 1)
        {
            while(head != nullptr)
            {
                Node* temp = head;
                head = head->next;
                delete temp;
            }
            return nullptr;
        }
        
        Node *ptr = head, *preptr = nullptr;
        while(ptr != nullptr)
        {
            int count = 1;
            while(count != K && ptr != nullptr)
            {
                preptr = ptr;
                ptr = ptr->next;
                count++;
            }
            
            if(ptr == nullptr) break;
            
            preptr->next = ptr->next;
            delete ptr;
            ptr = preptr->next;
        }
        return head;
    }
};

Q3. Rotate List (Leetcode 61)

class Solution {
public:
    ListNode* rotateRight(ListNode* head, int k) {
        if(head == nullptr || head->next == nullptr) return head;

        int count = 0;
        ListNode* temp = head;

        while(temp != nullptr)
        {
            count++;
            temp = temp->next;
        }

        k %= count;
        if(k == 0) return head;
        count -= k;

        ListNode *curr = head, *prev = nullptr;
        while(count--)
        {
            prev = curr;
            curr = curr->next;
        }
        prev->next = nullptr;

        ListNode *tail = curr;
        while(tail->next != nullptr) tail = tail->next;

        tail->next = head;
        return curr;
    }
};

Q3. Pallindrome Linked List (Leetcode 234)

/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        ListNode *prev = nullptr, *curr = head, *next = nullptr;

        while(curr != nullptr)
        {
            next = curr->next;
            curr->next = prev;
            prev = curr;
            curr = next;
        }

        return prev;
    }

    bool isPalindrome(ListNode* head) {
        int count = 0;
        ListNode* temp = head;
        ListNode* preptr = nullptr;

        while(temp != nullptr)
        {
            count++;
            temp = temp->next;
        }

        if(count == 1) return true;
        count /= 2;

        temp = head;
        while(count--)
        {
            preptr = temp;
            temp = temp->next;
        }
        preptr->next = nullptr; // 1->2 2->1

        temp = reverseList(temp); // 1->2 1->2

        while(head != nullptr && temp != nullptr)
        {
            if(head->val == temp->val)
            {
                head = head->next;
                temp = temp->next;
            }
            else return false;
        }
        return true;
    }
};
