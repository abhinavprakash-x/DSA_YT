Q1. Detect Loop in Linked List (GeekforGeeks)

class Solution {
  public:
    bool detectLoop(Node* head) {
        // code here
        if(head == nullptr || head->next == nullptr) return false;
        Node *slow = head, *fast = head->next;
        
        while(fast != nullptr && fast->next != nullptr)
        {
            slow = slow->next;
            fast = fast->next->next;
            
            if(slow == fast) return true;
        }
        return false;
    }
};

Q2. Cycle Length in Linked List (GeekforGeeks)

class Solution {
  public:
    int lengthOfLoop(Node *head) {
        // code here
        if(!head || !head->next) return 0;
        
        bool loop = false;
        int length = 0;
        Node *slow = head, *fast = head->next;
        
        while(fast && fast->next)
        {
            slow = slow->next;
            fast = fast->next->next;
            
            if(slow == fast)
            {
                loop = true;
                break;
            }
        }
        if(!loop) return 0;
        
        while(slow->next != fast)
        {
            length++;
            slow = slow->next;
        }
        
        return length + 1;
    }
};

Q3. Remove Cycle in Linked List (GeekforGeeks)

class Solution {
  public:
    void removeLoop(Node* head) {
        // code here
        if(!head || !head->next) return;
        Node *slow = head, *fast = head;
        bool loop = false;

        while(fast && fast->next)
        {
            slow = slow->next;
            fast = fast->next->next;

            if(slow == fast)
            {
                loop = true;
                break;
            }
        }
        if(!loop) return;
        
        slow = head;
        while(slow != fast)
        {
            slow = slow->next;
            fast = fast->next;
        }
        
        while(slow->next != fast) slow = slow->next;
        slow->next = nullptr;
    }
};

Q6. Split a Linked List Into Halves (GeekforGeeks)

class Solution {
  public:
    pair<Node *, Node *> splitList(struct Node *head) {
        if(head == nullptr || head->next == head)
                    return {head, nullptr};
        
        Node* slow = head;
        Node* fast = head;
        
        while(fast->next != head && fast->next->next != head)
        {
            slow = slow->next;
            fast = fast->next->next;
        }

        if(fast->next->next == head)
            fast = fast->next;
                    
        Node* head2 = slow->next;
        fast->next = head2;
        slow->next = head;
        
        return {head, head2};
    }
};

Q7. Insert in Sorted Circular Linked List (GeekforGeeks)

class Solution {
  public:
    Node* sortedInsert(Node* head, int data)
    {
        Node *temp = new Node(data);

        if(data <= head->data)
        {
            Node *ptr = head;

            while(ptr->next != head)
                ptr = ptr->next;

            ptr->next = temp;
            temp->next = head;
            return temp;
        }

        Node *ptr = head;

        while(ptr->next != head && ptr->next->data < data)
            ptr = ptr->next;

        temp->next = ptr->next;
        ptr->next = temp;

        return head;
    }
};

Q8. Intersection in Y Shaped Lists (GeekforGeeks)

class Solution {
  public:
    Node* intersectPoint(Node* head1, Node* head2)
    {
        int count1 = 0, count2 = 0;
        
        Node *ptr1 = head1;
        Node *ptr2 = head2;
        
        while(ptr1)
        {
            ptr1 = ptr1->next;
            count1++;
        }
        ptr1 = head1;
        
        while(ptr2)
        {
            ptr2 = ptr2->next;
            count2++;
        }
        ptr2 = head2;
        
        if(count1 < count2)
        {
            int fwd = count2 - count1;
            while(fwd--)
                ptr2 = ptr2->next;
        }
        else
        {
            int fwd = count1 - count2;
            while(fwd--)
                ptr1 = ptr1->next;
        }
        
        while(ptr1 && ptr2)
        {
            if(ptr1->next == ptr2->next)
                return ptr1->next;
                
            ptr1 = ptr1->next;
            ptr2 = ptr2->next;
        }
        
        return nullptr;
    }
};
