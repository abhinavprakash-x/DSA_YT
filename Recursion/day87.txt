Q1: Subset Sums (GeekforGeeks)

class Solution {
  public:
    void sum(vector<int>& arr, int idx, int n, int current_sum, vector<int>& ans) {
        if(idx == n) {
            ans.push_back(current_sum);
            return;
        }
        sum(arr, idx + 1, n, current_sum, ans);
        sum(arr, idx + 1, n, current_sum + arr[idx], ans);
    }
  
    vector<int> subsetSums(vector<int>& arr) {
        // code here
        vector<int> ans;
        sum(arr, 0, arr.size(), 0, ans);
        return ans;
    }
};

class Solution {

            if(0 == target) count++;
            return;
        }
        if(target < 0) return;
        
        sum(arr, target, idx + 1, n, count);
        sum(arr, target - arr[idx], idx + 1, n, count);
    }  

    int perfectSum(vector<int>& arr, int target) {

        sum(arr, target, 0, arr.size(), count);

    }
