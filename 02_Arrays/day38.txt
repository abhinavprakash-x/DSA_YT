1: Segregate 0's and 1's (GeekforGeeks)

class Solution {
  public:
    void segregate0and1(vector<int> &arr) {
        // code here
        int start = 0, end = arr.size() - 1;
        while(start < end) {
            while(start < end && arr[start] == 0) start++;
            while(start < end && arr[end] == 1) end--;
            swap(arr[start++], arr[end--]);
        }
    }
};

2: Two Sum II (Leetcode 167)

class Solution {
public:
    vector<int> twoSum(vector<int>& numbers, int target) {
        int start = 0, end = numbers.size() - 1;
        while(start < end) {
            if(numbers[start] + numbers[end] == target)
                return {start + 1, end + 1};
            else if(numbers[start] + numbers[end] < target) start++;
            else end--;
        }
        return {-1,-1};
    }
};

3: Pair With Given Difference (Interview Bit)

int Solution::solve(vector<int> &A, int B) {
    sort(A.begin(), A.end());
    int start = 0, end = 1, n = A.size();

    if(B < 0) B = -B;

    while(start < n && end < n) {
        if(start == end) {
            end++;
            continue;
        }

        int diff = A[end] - A[start];

        if(diff == B) return 1;
        else if(diff < B) end++;
        else start++;
    }

    return 0;
}

4: Product Pair (GeekforGeeks)

class Solution {
  public:
    bool isProduct(vector<int>& arr, long long target) {
        int n = arr.size();
        sort(arr.begin(), arr.end());

        for(int i = 0; i < n; i++) {

            // Handle zero separately
            if(arr[i] == 0) {
                if(target == 0) {
                    if(n > 1) return true;
                    continue;
                }
                continue;
            }

            if(target % arr[i] != 0) continue;

            long long query = target / arr[i];

            int start = 0, end = n - 1;

            while(start <= end) {
                int mid = start + (end - start) / 2;

                if(arr[mid] == query) {
                    if(mid != i) return true;
                    else {
                        // Check for another occurrence
                        if(mid > 0 && arr[mid - 1] == query) return true;
                        if(mid < n - 1 && arr[mid + 1] == query) return true;
                        break;
                    }
                }
                else if(arr[mid] < query)
                    start = mid + 1;
                else
                    end = mid - 1;
            }
        }

        return false;
    }
};

5: Remove Duplicates from Sorted Array (Interview Bit)

int Solution::removeDuplicates(vector<int> &A) {
    int n = A.size();
    int j = 0;
    for(int i = 1; i < n; i++) {
        if(A[i] != A[j]) {
            j++;
            A[j] = A[i];
        }
    }
    return j + 1;
}
