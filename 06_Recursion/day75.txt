Q1: Fibonacci Number (Leetcode 509)

class Solution {
public:
    int fib(int n) {
        if(n <= 1) return n;
        return fib(n - 1) + fib(n - 2);
    }
};

class Solution {
public:
    int climbStairs(int n) {
        if(n == 1) return 1;

        int prev2 = 1;
        int prev1 = 2;
        for(int i = 2; i < n; ++i) {
            int curr = prev2 + prev1;
            prev2 = prev1;
            prev1 = curr;
        }
        
        return prev1;
    }
};

        // code here

    }

Q4: Count ways with 3 moves (GeekforGeeks)

// Using DP
class Solution {
public:
    int countWays(int n) {
        // your code here
        // 1 = (1) -> 1
        // 2 = (2)(11) -> 2
        // 3 = (111)(12)(21)(3) -> 4
        // 4 = (1111)(112)(121)(211)(22)(31)(13) -> 7
        //
        // The pattern seems like f(n-1) + f(n-2) + f(n - 3)
        
        if(n == 1) return 1;
        if(n == 2) return 2;
        if(n == 3) return 4;

        int prev3 = 1;
        int prev2 = 2;
        int prev1 = 4;

        for(int i = 4; i <= n; ++i) {
            int curr = prev1 + prev2 + prev3;

            prev3 = prev2;
            prev2 = prev1;
            prev1 = curr;
        }

        return prev1;
    }
};

// Using Recursion
class Solution {
public:
    int countWays(int n) {
        if(n == 1) return 1;
        if(n == 2) return 2;
        if(n == 3) return 4;
        return countWays(n - 1) + countWays(n - 2) + countWays(n - 3);
    }
};

Q5: First n Fibonacci using DP

#define MOD 1000000007
class Solution {
  public:
    vector<int> fibonacciNumbers(int n) {
        // code here
        if(n == 1) return {0, 1};
        
        vector<int> ans(n + 1, 0);
        ans[0] = 0;
        ans[1] = 1;
        
        for(int i = 2; i <= n; ++i)
            ans[i] = (ans[i - 1] + ans[i - 2]) % MOD;
            
        return ans;
    }
};
// This isn't Recursion but it's honestly more efficient so why bother doing it Recursively
